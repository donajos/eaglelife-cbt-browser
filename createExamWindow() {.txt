function createExamWindow() {
    examWindow = new BrowserWindow({
        kiosk: true,
        frame: false,
        resizable: false,
        autoHideMenuBar: true,
        center: true,
        webPreferences: {
            nodeIntegration: false,
            contextIsolation: true,
            devTools: false
        }
    });

    // Prevent new windows and context menu
    examWindow.webContents.setWindowOpenHandler(() => ({ action: 'deny' }));
    examWindow.webContents.on('context-menu', e => e.preventDefault());

    // Block dangerous keyboard shortcuts
    examWindow.webContents.on('before-input-event', (event, input) => {
        if (
            input.key === 'F12' ||
            (input.control && input.shift && input.key.toLowerCase() === 'i') ||
            (input.control && input.key.toLowerCase() === 'r') ||
            (input.control && input.key.toLowerCase() === 'w') ||
            (input.alt && input.key === 'F4') ||
            (input.meta && input.key.toLowerCase() === 'q')
        ) {
            event.preventDefault();
        }
    });

    // Restrict navigation to the server IP only
    const blockExternal = (event, url) => {
        try {
            const parsed = new URL(url);
            if (parsed.hostname !== serverIP) {
                event.preventDefault();
            }
        } catch {
            event.preventDefault();
        }
    };
    examWindow.webContents.on('will-navigate', blockExternal);
    examWindow.webContents.on('will-redirect', blockExternal);

    // Load the exam page
    const examURL = `http://${serverIP}/cbtexam/login/`;
    lastExamURL = examURL;
    examWindow.loadURL(examURL);

    examWindow.webContents.on('did-start-navigation', (event, url, isInPlace, isMainFrame) => {
        if (isMainFrame) {
            lastExamURL = url;
        }
    });
    

    // Retry if loading fails
    examWindow.webContents.on('did-fail-load', (e, code, desc, url, isMainFrame) => {
        if (!isMainFrame || code === -3) return;
        
        console.log("Page failed:", desc);

        // Show overlay in page
        examWindow.webContents.executeJavaScript(`showOfflineOverlay && showOfflineOverlay();`);

        if (!retryInterval) {
            retryInterval = setInterval(async () => {
                try {
                    await checkServer(serverIP, 80); // check if server is back

                    clearInterval(retryInterval);
                    retryInterval = null;

                    if (examWindow && !examWindow.isDestroyed()) {
                        examWindow.loadURL(lastExamURL || `http://${serverIP}/cbtexam/my/`);
                    }
                } catch {
                // server still unreachable, keep waiting
                }
            }, 5000);
        }
    });



    // Offline/online detection with overlay
    examWindow.webContents.on('did-finish-load', () => {

        if (retryInterval) {
        clearInterval(retryInterval);
        retryInterval = null;
        }

        
    });

    //const { net } = require('electron');

    
   
    
    reconnectInterval = setInterval(async () => {
        if (!examWindow || examWindow.isDestroyed()) return;
        
        try {
            await checkServer(serverIP, 80);
            
            const url = examWindow.webContents.getURL();
            if (!url || url.startsWith("chrome-error://") || url === "about:blank") {
                const targetURL = lastExamURL || `http://${serverIP}/cbtexam/my/`;
                examWindow.loadURL(targetURL);
            }
        } catch {
        // still offline
        }
    }, 5000);

    examWindow.webContents.on('will-redirect', (event, url) => {
        lastExamURL = url;
    });

    examWindow.on('blur', () => {
        if (examWindow && !examWindow.isDestroyed()) {
        examWindow.focus();
      }
    });

    /* examWindow.on('minimize', (e) => {
        e.preventDefault();
    }); */


    examWindow.on('closed', () => {
        if (retryInterval) {
            clearInterval(retryInterval);
            retryInterval = null;
        }
        if (reconnectInterval) {
            clearInterval(reconnectInterval);
            reconnectInterval = null;
        }
        examWindow = null;

        // reopen exam window automatically
        setTimeout(() => {
            if (!examWindow) {
                createExamWindow();
            }
        }, 1000);
    });
    
}
